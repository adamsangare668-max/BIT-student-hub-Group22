TP1 — Gestion d'un tableau d'entiers

Énoncé : Programme avec menu : saisie, affichage, moyenne, suppression du max, suppression du min, ajout à une position, quitter.

#include <stdio.h>
#include <stdlib.h>
#define NMAX 20

int tab[NMAX], nb = 0;

void saisie() {
    printf("Nombre d'éléments (max %d) : ", NMAX);
    scanf("%d", &nb);
    for(int i=0; i<nb; i++) {
        printf("tab[%d] = ", i);
        scanf("%d", &tab[i]);
    }
}

void affichage() {
    for(int i=0; i<nb; i++) printf("%d ", tab[i]);
    printf("\n");
}

float moyenne() {
    float s = 0;
    for(int i=0; i<nb; i++) s += tab[i];
    return s/nb;
}

int pos_max() {
    int pmax = 0;
    for(int i=1; i<nb; i++) if(tab[i] > tab[pmax]) pmax = i;
    return pmax;
}

int pos_min() {
    int pmin = 0;
    for(int i=1; i<nb; i++) if(tab[i] < tab[pmin]) pmin = i;
    return pmin;
}

void supprimer_pos(int p) {
    for(int i=p; i<nb-1; i++) tab[i] = tab[i+1];
    nb--;
}

void ajouter() {
    int val, pos;
    if(nb == NMAX) { printf("Tableau plein\n"); return; }
    printf("Valeur à ajouter : "); scanf("%d", &val);
    printf("Position (0..%d) : ", nb); scanf("%d", &pos);
    if(pos < 0 || pos > nb) pos = nb;
    for(int i=nb; i>pos; i--) tab[i] = tab[i-1];
    tab[pos] = val;
    nb++;
}

int main() {
    char choix;
    do {
        printf("\nA-Saisie/affichage B-Moyenne C-Suppr Max D-Suppr Min E-Ajout Q-Quitter\n");
        choix = getchar(); getchar(); // pour vider le buffer
        switch(choix) {
            case 'A': saisie(); affichage(); break;
            case 'B': printf("Moyenne = %.2f\n", moyenne()); break;
            case 'C': supprimer_pos(pos_max()); affichage(); break;
            case 'D': supprimer_pos(pos_min()); affichage(); break;
            case 'E': ajouter(); affichage(); break;
            case 'Q': printf("Fin\n"); break;
            default: printf("Choix invalide\n");
        }
    } while(choix != 'Q');
    return 0;
}

TP2 — Gestion d'enseignants (structures)

Énoncé : Tableau d'enseignants avec nom, prénom, heures par matière. Menu : saisie, construction, modification, tri, quitter.

#include <stdio.h>
#include <string.h>
#define NM_max 3
#define NE_max 20

struct Enseignant {
    char nom_pren[40];
    char nom[20];
    char pren[20];
    int NH[NM_max];
};

struct Enseignant ens[NE_max];
int nb = 0;

void saisie() {
    printf("Nombre d'enseignants : "); scanf("%d", &nb);
    for(int i=0; i<nb; i++) {
        printf("Nom : "); scanf("%s", ens[i].nom);
        printf("Prénom : "); scanf("%s", ens[i].pren);
        strcpy(ens[i].nom_pren, ens[i].nom);
        strcat(ens[i].nom_pren, " ");
        strcat(ens[i].nom_pren, ens[i].pren);
        for(int j=0; j<NM_max; j++) {
            printf("Heures matière %d : ", j+1);
            scanf("%d", &ens[i].NH[j]);
        }
    }
}

void affichage() {
    for(int i=0; i<nb; i++) {
        printf("%s : ", ens[i].nom_pren);
        for(int j=0; j<NM_max; j++) printf("%d ", ens[i].NH[j]);
        printf("\n");
    }
}

TP3 — Liste simplement chaînée

Énoncé : Créer une liste avec insertion en tête, sauvegarde dans fichier, insertion triée, libération mémoire.

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};
typedef struct Node Cell;

Cell *debut = NULL;

void push(int val) {
    Cell *p = (Cell*)malloc(sizeof(Cell));
    p->data = val;
    p->next = debut;
    debut = p;
}

void affiche() {
    Cell *p = debut;
    while(p) { printf("%d ", p->data); p = p->next; }
    printf("\n");
}

void sauvegarde(char *nom) {
    FILE *f = fopen(nom, "w");
    Cell *p = debut;
    while(p) { fprintf(f, "%d\n", p->data); p = p->next; }
    fclose(f);
}

void insertion_triee(int val) {
    // Insère en préservant l'ordre croissant
    Cell *p = (Cell*)malloc(sizeof(Cell));
    p->data = val;
    if(!debut || debut->data > val) { p->next = debut; debut = p; return; }
    Cell *q = debut;
    while(q->next && q->next->data < val) q = q->next;
    p->next = q->next;
    q->next = p;
}

void liberer() {
    Cell *p;
    while(debut) { p = debut; debut = debut->next; free(p); }
}

int main() {
    int n;
    printf("n = "); scanf("%d", &n);
    for(int i=n; i>=0; i-=2) push(i);
    affiche();
    sauvegarde("Liste1");
    for(int i=-1; i<=2*n; i+=2) insertion_triee(i);
    affiche();
    sauvegarde("Liste2");
    liberer();
    return 0;
}

TP — Liste doublement chaînée

Énoncé : Implémenter une liste doublement chaînée avec insertion en tête, en queue, insertion à une position, suppression.

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *prev;
    struct Node *next;
};
typedef struct Node Cell;

Cell *debut = NULL, *fin = NULL;

void push(int val) {
    Cell *p = (Cell*)malloc(sizeof(Cell));
    p->data = val;
    p->prev = NULL;
    p->next = debut;
    if(debut) debut->prev = p;
    else fin = p;
    debut = p;
}

void push_back(int val) {
    Cell *p = (Cell*)malloc(sizeof(Cell));
    p->data = val;
    p->next = NULL;
    p->prev = fin;
    if(fin) fin->next = p;
    else debut = p;
    fin = p;
}

void affiche() {
    Cell *p = debut;
    while(p) { printf("%d ", p->data); p = p->next; }
    printf("\n");
}

void liberer() {
    Cell *p;
    while(debut) { p = debut; debut = debut->next; free(p); }
    fin = NULL;
}

TP — Allocation dynamique

Énoncé : Créer un tableau de taille quelconque avec malloc, le remplir, l'afficher, le libérer.

#include <stdio.h>
#include <stdlib.h>

int main() {
    int dim;
    printf("Taille du tableau : "); scanf("%d", &dim);
    int *tab = (int*)malloc(dim * sizeof(int));
    if(!tab) { printf("Erreur mémoire\n"); return 1; }
    for(int i=0; i<dim; i++) tab[i] = i*i;
    for(int i=0; i<dim; i++) printf("%d ", tab[i]);
    free(tab);
    return 0;
}

TP — Lecture/Écriture fichier

Énoncé : Sauvegarder un tableau dans un fichier et le relire.

#include <stdio.h>
#define N 10

void sauvegarde(float x[], int ind[]) {
    FILE *f = fopen("monfichier.dat", "w");
    if(f) {
        for(int i=0; i<N; i++) fprintf(f, "%f %d\n", x[i], ind[i]);
        fclose(f);
    }
}

void lecture(float x[], int ind[]) {
    FILE *f = fopen("monfichier.dat", "r");
    if(f) {
        int i=0;
        while(!feof(f) && i<N) {
            fscanf(f, "%f %d", &x[i], &ind[i]);
            i++;
        }
        fclose(f);
    }
}
